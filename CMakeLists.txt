cmake_minimum_required(VERSION 3.24)
project(Air2Win VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network Test)
qt_standard_project_setup(REQUIRES 6.8)

add_library(receiver STATIC src/ReceiverController.cpp src/ReceiverController.h)
target_include_directories(receiver PUBLIC src)
target_link_libraries(receiver PUBLIC Qt6::Core Qt6::Network)
target_compile_definitions(receiver PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(receiver PRIVATE /W4 /utf-8)
else()
    target_compile_options(receiver PRIVATE -Wall -Wextra -Wpedantic)
endif()

qt_add_executable(Air2Win WIN32 src/main.cpp)
if(WIN32)
    configure_file(src/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
    target_sources(Air2Win PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
endif()
qt_add_resources(Air2Win branding PREFIX "/branding" BASE docs FILES docs/7629289.png)
target_link_libraries(Air2Win PRIVATE receiver Qt6::Quick Qt6::QuickControls2)
qt_add_qml_module(Air2Win URI Air2Win VERSION 1.0
    QML_FILES qml/Main.qml qml/Icon.qml qml/ActionButton.qml)
set_target_properties(Air2Win PROPERTIES OUTPUT_NAME Air2Win)

include(CTest)
if(BUILD_TESTING)
    qt_add_executable(fake_receiver tests/fake_receiver.cpp)
    target_link_libraries(fake_receiver PRIVATE Qt6::Core)
    qt_add_executable(receiver_tests tests/receiver_tests.cpp)
    target_link_libraries(receiver_tests PRIVATE receiver Qt6::Test)
    add_test(NAME receiver_lifecycle COMMAND receiver_tests $<TARGET_FILE:fake_receiver>)
endif()

install(TARGETS Air2Win RUNTIME DESTINATION .)
